/**
 * 🚀 EXPLAINING API REQUESTS (GET vs POST)
 * Think of an API like a waiter in a restaurant:
 * - GET: You ask for the menu (Fetching information).
 * - POST: You give your order to the waiter (Sending information).
 */

// 1. THE GET REQUEST (Retrieve Data)
// Purpose: Use this when you want to "GET" something from the database.

const getEmployees = async () => {
  try {
    // By default, apiCall uses 'GET' if you don't specify a method
    const response = await apiCall('/employees');
    const result = await response.json();

    if (result.status === 'success') {
      console.log("Here is the list of employees:", result.data);
    }
  } catch (error) {
    console.error("The GET request failed:", error);
  }
};


// 2. THE POST REQUEST (Send/Create Data)
// Purpose: Use this when you want to "POST" (send) new data to the server.
let a = {
  "tet": "test",
  "hihhh": "nknkn",
  "knknn": 10
}
const addNewEmployee = async () => {
  const newEmployeeData = {
    name: "John Doe",
    role: "Software Engineer",
    email: "john@example.com"
  };

  try {
    const response = await apiCall('/employee-manage', {
      method: 'POST', // You MUST specify 'POST'
      body: JSON.stringify(newEmployeeData), // Data must be turned into a string
    });

    const result = await response.json();

    if (result.status === 'success') {
      alert("Employee added successfully!");
    }
  } catch (error) {
    console.error("The POST request failed:", error);
  }
};

/**
 * 💡 KEY DIFFERENCES:
 * 
 * GET: 
 * - No 'body' allowed.
 * - Used for reading.
 * - Like looking at a profile.
 * 
 * POST: 
 * - Has a 'body' containing your data.
 * - Used for creating/updating.
 * - Like submitting a sign-up form.
 */
